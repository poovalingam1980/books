const CEOSVG_FOLDER_ID = '1w1nM7tekSvSM374CPgcH6_AY1meSZm1z';
const DIGITALBOOKS_FOLDER_ID = '1QXOrMIqF_eFaWeWgLFAGzlgvPkqAyNjA';
const SPREADSHEET_ID = '1KOBO0_plHmDPz5kN2BhzVJfZx1F69aMIaKO50_SlN0E';

function doGet(e) {
  try {
    const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'status';
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    if (action === 'getBooks') {
      const sheet = ss.getSheetByName('Books');
      const rows = sheet.getDataRange().getValues();
      const headers = rows[0];
      const books = rows.slice(1).map(r => {
        const obj = {};
        headers.forEach((h, i) => obj[h] = r[i]);
        return obj;
      });
      return jsonResponse({ status: 'success', books: books });
    }

    if (action === 'getUsers') {
      const sheet = ss.getSheetByName('Users');
      const rows = sheet.getDataRange().getValues();
      const headers = rows[0];
      const users = rows.slice(1).map(r => {
        const obj = {};
        headers.forEach((h, i) => obj[h] = r[i]);
        return obj;
      });
      return jsonResponse({ status: 'success', users: users });
    }

    if (action === 'getHtmlContent') {
      const fileId = (e && e.parameter && e.parameter.fileId) ? e.parameter.fileId : '';
      if (!fileId) return jsonResponse({ status: 'error', message: 'Missing fileId' });
      const file = DriveApp.getFileById(fileId);
      const htmlStr = file.getBlob().getDataAsString('UTF-8');
      return jsonResponse({ status: 'success', fileId: fileId, htmlContent: htmlStr });
    }

    return jsonResponse({
      status: 'success',
      message: 'சிவகங்கை கல்விக்களம் (library.ceosvg.in) Backend API இயங்குகிறது!'
    });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

function doPost(e) {
  try {
    const rawData = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const payload = JSON.parse(rawData);
    const action = payload.action;
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const nowStr = Utilities.formatDate(new Date(), 'Asia/Kolkata', 'yyyy-MM-dd HH:mm:ss');

    if (action === 'sendEmailOtp') {
      const email = String(payload.email || '').trim();
      const otp = String(payload.otp || '');
      const purpose = payload.purpose === 'RESET' ? 'கடவுச்சொல் மீட்டமைப்பு (Password Reset)' : 'புதிய பயனர் கணக்கு பதிவு (Account Registration)';

      if (!email || !otp) {
        return jsonResponse({ status: 'error', message: 'மின்னஞ்சல் மற்றும் OTP விவரம் தேவை!' });
      }

      const subject = 'சிவகங்கை கல்விக்களம் (library.ceosvg.in) - OTP சரிபார்ப்பு எண்: ' + otp;
      const htmlBody = '<div style="font-family:sans-serif;max-width:520px;margin:auto;padding:24px;border:1px solid #e2e8f0;border-radius:16px;">' +
        '<h2 style="color:#312e81;margin-top:0;">சிவகங்கை கல்விக்களம் (library.ceosvg.in)</h2>' +
        '<p>வணக்கம் <strong>' + (payload.fullName || payload.userId || 'பயனர்') + '</strong>,</p>' +
        '<p>உங்கள் <strong>' + purpose + '</strong> சரிபார்ப்புக்கான ஒருமுறைக் கடவுச்சொல் (OTP):</p>' +
        '<div style="font-size:28px;font-weight:800;letter-spacing:6px;color:#4338ca;background:#eef2ff;padding:14px 20px;border-radius:12px;text-align:center;margin:16px 0;">' +
        otp + '</div>' +
        '<p style="font-size:12px;color:#64748b;">கைபேசி எண் (User ID): ' + (payload.userId || '') + '</p>' +
        '</div>';

      MailApp.sendEmail({
        to: email,
        subject: subject,
        htmlBody: htmlBody
      });

      return jsonResponse({
        status: 'success',
        message: 'OTP உங்கள் மின்னஞ்சலுக்கு (' + email + ') அனுப்பப்பட்டது!'
      });
    }

    if (action === 'registerUser') {
      let userSheet = ss.getSheetByName('Users');
      if (!userSheet) {
        userSheet = ss.insertSheet('Users');
        userSheet.appendRow(['User_ID_Mobile', 'Password', 'Full_Name', 'Role', 'Email', 'Status', 'Created_At']);
      }

      const mobileId = String(payload.userId || '').trim();
      if (!/^[0-9]{10}$/.test(mobileId)) {
        return jsonResponse({ status: 'error', message: 'பயனர் ID 10 இலக்க கைபேசி எண்ணாக மட்டுமே இருக்க வேண்டும்!' });
      }

      const data = userSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim() === mobileId) {
          return jsonResponse({ status: 'error', message: 'இந்த கைபேசி எண் ஏற்கனவே பதிவு செய்யப்பட்டுள்ளது!' });
        }
      }

      userSheet.appendRow([
        mobileId,
        payload.password,
        payload.fullName || mobileId,
        payload.role || 'மாணவர் (Student)',
        payload.email || '',
        'Active (OTP Verified)',
        nowStr
      ]);

      return jsonResponse({
        status: 'success',
        message: 'பயனர் விவரம் மற்றும் கடவுச்சொல் Google Sheet-ல் சேமிக்கப்பட்டது!'
      });
    }

    if (action === 'resetPassword') {
      const userSheet = ss.getSheetByName('Users');
      if (!userSheet) {
        return jsonResponse({ status: 'error', message: 'Users தாள் கிடைக்கவில்லை!' });
      }
      const mobileId = String(payload.userId || '').trim();
      const email = String(payload.email || '').trim().toLowerCase();
      const newPassword = String(payload.newPassword || '').trim();

      const data = userSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        const rowMobile = String(data[i][0]).trim();
        const rowEmail = String(data[i][4]).trim().toLowerCase();
        if (rowMobile === mobileId && rowEmail === email) {
          userSheet.getRange(i + 1, 2).setValue(newPassword);
          return jsonResponse({
            status: 'success',
            message: 'உங்கள் புதிய கடவுச்சொல் Google Sheet-ல் வெற்றிகரமாகப் புதுப்பிக்கப்பட்டது!'
          });
        }
      }
      return jsonResponse({ status: 'error', message: 'கைபேசி எண் மற்றும் மின்னஞ்சல் பொருந்தவில்லை!' });
    }

    if (action === 'loginUser') {
      const userSheet = ss.getSheetByName('Users');
      const data = userSheet.getDataRange().getValues();
      const inputId = String(payload.userId || '').trim();
      const inputPwd = String(payload.password || '').trim();

      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim() === inputId && String(data[i][1]).trim() === inputPwd) {
          return jsonResponse({
            status: 'success',
            user: {
              userId: String(data[i][0]).trim(),
              fullName: data[i][2],
              role: data[i][3],
              email: data[i][4]
            }
          });
        }
      }
      return jsonResponse({ status: 'error', message: 'தவறான கைபேசி எண் அல்லது கடவுச்சொல்!' });
    }

    if (action === 'uploadBook') {
      const digitalBooksFolder = DriveApp.getFolderById(DIGITALBOOKS_FOLDER_ID);
      let bookSheet = ss.getSheetByName('Books');
      if (!bookSheet) {
        bookSheet = ss.insertSheet('Books');
        bookSheet.appendRow([
          'Book_ID', 'Title_Tamil', 'Title_English', 'Class_Level', 'Medium',
          'Formats', 'HTML_File_URL', 'PDF_File_URL', 'Flipbook_URL',
          'Cover_Image_URL', 'Uploaded_By', 'Uploaded_At'
        ]);
      }

      const safeTitle = (payload.titleEn || payload.titleTa || 'Book').replace(/[^a-zA-Z0-9_-]/g, '_');
      let htmlDriveUrl = payload.htmlExternalUrl || '';
      let pdfDriveUrl = payload.pdfExternalUrl || '';
      let flipDriveUrl = payload.flipbookExternalUrl || '';
      let coverDriveUrl = '';

      if (payload.pdfBase64) {
        const pdfFile = saveBase64ToFolder(
          digitalBooksFolder,
          payload.pdfFileName || (safeTitle + '.pdf'),
          payload.pdfBase64,
          'application/pdf'
        );
        pdfDriveUrl = pdfFile.getUrl();
        if (!flipDriveUrl) flipDriveUrl = pdfDriveUrl;
      }

      if (payload.htmlContent) {
        const htmlBlob = Utilities.newBlob(payload.htmlContent, 'text/html', safeTitle + '.html');
        const htmlFile = digitalBooksFolder.createFile(htmlBlob);
        htmlFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        htmlDriveUrl = htmlFile.getUrl();
      }

      if (payload.coverBase64 && payload.coverBase64.indexOf('data:image') === 0) {
        const coverFile = saveBase64ToFolder(
          digitalBooksFolder,
          safeTitle + '_cover.jpg',
          payload.coverBase64,
          'image/jpeg'
        );
        coverDriveUrl = coverFile.getUrl();
      }

      if (Array.isArray(payload.flipbookImagesBase64) && payload.flipbookImagesBase64.length > 0) {
        const flipSubFolder = digitalBooksFolder.createFolder(safeTitle + '_Flipbook_Pages');
        flipSubFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        payload.flipbookImagesBase64.forEach((imgB64, idx) => {
          saveBase64ToFolder(flipSubFolder, 'page_' + (idx + 1) + '.jpg', imgB64, 'image/jpeg');
        });
        flipDriveUrl = flipSubFolder.getUrl();
      }

      const bookId = payload.id || ('book-' + new Date().getTime());
      bookSheet.appendRow([
        bookId,
        payload.titleTa || '',
        payload.titleEn || '',
        payload.classLevel || 'Class XII',
        payload.medium || 'இருமொழி',
        payload.formatsList || 'HTML',
        htmlDriveUrl || ('https://drive.google.com/drive/folders/' + DIGITALBOOKS_FOLDER_ID),
        pdfDriveUrl || ('https://drive.google.com/drive/folders/' + DIGITALBOOKS_FOLDER_ID),
        flipDriveUrl || ('https://drive.google.com/drive/folders/' + DIGITALBOOKS_FOLDER_ID),
        coverDriveUrl || 'Default Cover',
        payload.uploadedBy || '9999999999',
        nowStr
      ]);

      return jsonResponse({
        status: 'success',
        message: 'புத்தகம் digitalbooks கோப்புறையிலும் Google Sheet-லும் வெற்றிகரமாகச் சேமிக்கப்பட்டது!',
        bookId: bookId
      });
    }

    return jsonResponse({ status: 'error', message: 'Unknown action: ' + action });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

function saveBase64ToFolder(folder, fileName, base64Data, defaultMimeType) {
  let cleanBase64 = base64Data;
  let mimeType = defaultMimeType || 'application/octet-stream';
  if (base64Data.indexOf(',') !== -1) {
    const parts = base64Data.split(',');
    const header = parts[0];
    cleanBase64 = parts[1];
    const mimeMatch = header.match(/data:([^;]+);base64/);
    if (mimeMatch && mimeMatch[1]) mimeType = mimeMatch[1];
  }
  const decodedBytes = Utilities.base64Decode(cleanBase64);
  const blob = Utilities.newBlob(decodedBytes, mimeType, fileName);
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file;
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
